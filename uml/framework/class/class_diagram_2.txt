' ========= Интерфейсы (дублируются для контекста) =========
interface Preparator {
  + prepare(rawData: Any) : List<ProcessorInput>
}

interface Processor {
  + process(input: List<ProcessorInput>) : List<ProcessorOutput>
}

interface Interpreter {
  + interpret(output: List<ProcessorOutput>) : Any
}

interface Trainer {
  + train(input: List<ProcessorInput>) : void
}

' ========= Пайплайны =========
class TrainPipeline {
  - preparator: Preparator
  - trainer: Trainer
  ..
  + run(rawData: Any) : void
  + setPreparator(Preparator) : void
  + setTrainer(Trainer) : void
}

class InferencePipeline {
  - preparator: Preparator
  - processor: Processor
  - interpreter: Interpreter
  ..
  + run(rawData: Any) : Any
  + setPreparator(Preparator) : void
  + setProcessor(Processor) : void
  + setInterpreter(Interpreter) : void
}

' ========= Связи =========
TrainPipeline *-- Preparator
TrainPipeline *-- Trainer

InferencePipeline *-- Preparator
InferencePipeline *-- Processor
InferencePipeline *-- Interpreter

note bottom of TrainPipeline
  <b>TrainPipeline.run(rawData):</b>
  1. List<ProcessorInput> = preparator.prepare(rawData)
  2. trainer.train(весь список)
  
  Пайплайн передаёт список целиком.
  Trainer сам решает, как обрабатывать
  (батчами, целиком, фильтровать).
end note

note bottom of InferencePipeline
  <b>InferencePipeline.run(rawData):</b>
  1. List<ProcessorInput> = preparator.prepare(rawData)
  2. List<ProcessorOutput> = processor.process(весь список)
  3. Any = interpreter.interpret(весь список результатов)
  
  Пайплайн передаёт списки целиком.
  Интерпретатор получает все результаты сразу
  для агрегированной визуализации.
end note

note top of Processor
  Модель загружается из файла
  внутри process().
  Фреймворк не управляет этим.
  
  Принимает и возвращает списки.
  Для одного элемента — список из 1.
end note

note top of Trainer
  train() сохраняет модель
  в файл (pickle, H5, ONNX...).
  Фреймворк не управляет форматом.
  
  Получает список ProcessorInput.
  Для одного элемента — список из 1.
end note

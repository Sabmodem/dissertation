@startuml
!theme plain

skinparam componentBorderColor DarkSlateGray
skinparam interfaceBackgroundColor LightCyan
skinparam interfaceBorderColor SteelBlue

package "Ядро Фреймворка" as Core {
  
  package "Контракты (DTO)" as DTO {
    component "ProcessorInput" as PI <<abstract>>
    component "ProcessorOutput" as PO <<abstract>>
  }
  
  package "Интерфейсы" as Interfaces {
    interface "Preparator" as PrepI
    interface "Processor" as ProcI
    interface "Interpreter" as InterpI
    interface "Trainer" as TrainI
  }
  
  package "Пайплайны" as Pipelines {
    component "TrainPipeline" as TrainP
    component "InferencePipeline" as InfP
  }
}

' ========= Связи внутри фреймворка =========
PrepI ..> PI : <<создает>>
ProcI ..> PI : <<использует>>
ProcI ..> PO : <<создает>>
TrainI ..> PI : <<использует>>
InterpI ..> PO : <<использует>>

TrainP .. PrepI : <<requires>>
TrainP .. TrainI : <<requires>>

InfP .. PrepI : <<requires>>
InfP .. ProcI : <<requires>>
InfP .. InterpI : <<requires>>

' ========= Примечания =========
note right of TrainP
  <b>TrainPipeline</b>
  Владеет: Preparator, Trainer
  Выход: файл модели
end note

note right of InfP
  <b>InferencePipeline</b>
  Владеет: Preparator, Processor, Interpreter
  Выход: результат интерпретации
end note

note bottom of DTO
  <b>Точки расширения</b>
  Пользователь обязан унаследовать
  ProcessorInput и ProcessorOutput
end note

note bottom of Interfaces
  <b>Обязательные интерфейсы</b>
  Все 4 должны быть реализованы
  пользователем
end note

@enduml

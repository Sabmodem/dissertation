@startuml
!theme plain
skinparam classBackgroundColor LightYellow
skinparam classBorderColor DarkSlateGray
skinparam interfaceBackgroundColor LightCyan
skinparam interfaceBorderColor SteelBlue
skinparam abstractClassBackgroundColor Lavender

' ========= Контракты данных (Extension Points) =========
abstract class ProcessorInput <<DTO>> {
}

abstract class ProcessorOutput <<DTO>> {
}

' ========= Интерфейсы этапов =========
interface Preparator {
  + prepare(rawData: Any) : ProcessorInput
}

interface Processor {
  + process(input: ProcessorInput) : ProcessorOutput
}

interface Interpreter {
  + interpret(output: ProcessorOutput) : Any
}

interface Trainer {
  + train(input: ProcessorInput) : void
}

' ========= Пайплайны =========
class TrainPipeline {
  - preparator: Preparator
  - trainer: Trainer
  ..
  + run(rawData: Any) : void
  + setPreparator(Preparator) : void
  + setTrainer(Trainer) : void
}

class InferencePipeline {
  - preparator: Preparator
  - processor: Processor
  - interpreter: Interpreter
  ..
  + run(rawData: Any) : Any
  + setPreparator(Preparator) : void
  + setProcessor(Processor) : void
  + setInterpreter(Interpreter) : void
}

' ========= Связи =========
TrainPipeline *-- Preparator
TrainPipeline *-- Trainer

InferencePipeline *-- Preparator
InferencePipeline *-- Processor
InferencePipeline *-- Interpreter

Preparator ..> ProcessorInput : <<creates>>
Processor ..> ProcessorInput : <<uses>>
Processor ..> ProcessorOutput : <<creates>>
Trainer ..> ProcessorInput : <<uses>>
Interpreter ..> ProcessorOutput : <<uses>>

' Примечание о модели
note top of Processor
  Модель загружается из файла
  внутри process().
  Фреймворк не управляет этим.
end note

note top of Trainer
  train() сохраняет модель
  в файл (pickle, H5, ONNX...).
  Фреймворк не управляет форматом.
end note

note top of ProcessorInput
  <b>Точки расширения фреймворка</b>
  Пользователь <b>обязан</b> унаследовать
  ProcessorInput и ProcessorOutput
end note

@enduml

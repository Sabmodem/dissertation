package "ML Pipeline Framework (точки расширения)" as Framework {
    abstract class ProcessorInput <<DTO>> {
    }
    abstract class ProcessorOutput <<DTO>> {
    }

    interface Preparator {
        + prepare(rawData: Any) : ProcessorInput
    }

    interface Trainer {
        + train(input: ProcessorInput) : void
    }

    interface Processor {
        + process(input: ProcessorInput) : ProcessorOutput
    }

    interface Interpreter {
        + interpret(output: ProcessorOutput) : Any
    }
}

' ============================================================
' ПОЛЬЗОВАТЕЛЬСКИЕ DTO
' ============================================================
package "Data Transfer Objects" as DTOs {
    class FraudInput extends ProcessorInput {
        - X: np.ndarray
        - y: np.ndarray
        - feature_names: List[str]
        - dataset_name: str
    }

    class FraudOutput extends ProcessorOutput {
        - y_true: np.ndarray
        - y_pred: np.ndarray
        - y_proba: np.ndarray
        - metrics: Dict[str, float]
        - confusion_matrix: np.ndarray
        - roc_curve: Dict
        - pr_curve: Dict
    }
}

' ============================================================
' АБСТРАКЦИИ ПАЙПЛАЙНА
' ============================================================
package "Core Abstractions" as Core {
    abstract class BaseModel {
        # model: Any
        # is_trained: bool
        # config: Dict
        ..
        + build_model() : void
        + train(X_train, y_train, X_val?, y_val?) : Dict
        + predict(X) : np.ndarray
        + predict_proba(X) : np.ndarray
        + save_model(filepath) : void
        + load_model(filepath) : void
        + get_config() : Dict
    }

    abstract class BaseVectorizer {
        # vectorizer: Any
        # is_fitted: bool
        # config: Dict
        ..
        + fit(texts) : BaseVectorizer
        + transform(texts) : np.ndarray
        + fit_transform(texts) : np.ndarray
        + get_feature_names() : List[str]
        + get_output_dim() : int
        - _get_output_dim() : int
    }
}

note top of BaseModel
    <b>Архитектурное решение:</b>
    BaseModel совмещает Trainer и Processor.
    Это оправдано для Fraud Detection,
    где модель = самодостаточный артефакт.
    train() → Trainer.train()
    process() → Processor.process()
end note

note top of BaseVectorizer
    <b>Архитектурное решение:</b>
    BaseVectorizer реализует Preparator.
    prepare() принимает сырые тексты,
    возвращает FraudInput с векторизованными данными.
end note

note top of Experiment
    <b>Оркестратор поверх фреймворка:</b>
    Автоматизирует полный цикл:
    DataLoader → Model.train() → Evaluator → Visualizer.
    Фреймворк такого не требует,
    это добавленная автоматизация.
end note

note bottom of Evaluator
    <b>Расширение фреймворка:</b>
    Evaluator не имеет аналога во фреймворке.
    Добавлен для стандартизации метрик
    и удобства сравнения моделей.
end note

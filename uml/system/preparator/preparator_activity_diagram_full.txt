@startuml
!theme plain
skinparam activityBackgroundColor LightYellow
skinparam activityBorderColor DarkSlateGray

start

:Вход: rawData — список CSV-строк\nили DataFrame с полями\n"Fillings" (текст MD&A) и "Fraud" (метка);

:Извлечение текстов и меток\nиз rawData;

if (fit_mode?) then (<b>True (Обучение)</b>)
  :Инициализировать новый\nTfidfVectorizer и StandardScaler;
else (<b>False (Инференс)</b>)
  :Загрузить TfidfVectorizer\nи StandardScaler с диска\n(из artifact_path);
endif

:Поочерёдная обработка каждого документа;

partition "Очистка текста (TextCleaner)" {
  :Удаление HTML-тегов\n(Cleaner.remove_html_tags);
  :Удаление служебных заголовков\n(Cleaner.remove_headers);
  :Удаление юридических оговорок\n(Cleaner.remove_legal_disclaimers);
  :Нормализация пробелов\n(Cleaner.normalize_whitespace);
}

note right
  <b>Важно:</b> цифры НЕ удаляются.
  Стоп-слова удаляются выборочно —
  юридические термины сохраняются.
end note

partition "Извлечение лингвистических признаков" {
  :Вычисление Word Count;
  :Вычисление Fog Index;
  :Вычисление Flesch Reading Ease;
  :Вычисление доли пассивных конструкций;
  :Вычисление тональности\n(словарь Loughran-McDonald);
  :Вычисление доли слов неопределённости;
  
  :Формирование словаря\nlinguistic_features;
}

:Накопление очищенных текстов\nв список;

:Обработка всех документов завершена;

if (fit_mode?) then (<b>True (Обучение)</b>)
  :Обучить TfidfVectorizer\nна корпусе очищенных текстов;
  :Трансформировать корпус →\nматрица TF-IDF;
else (<b>False (Инференс)</b>)
  :Трансформировать корпус\nобученным TfidfVectorizer'ом;
endif

:Объединение TF-IDF матрицы\nи лингвистических признаков\nв combined_features;

if (fit_mode?) then (<b>True (Обучение)</b>)
  :Обучить StandardScaler\nна combined_features;
  :Сохранить артефакты на диск:\n• tfidf_vectorizer.pkl\n• feature_scaler.pkl;
  :Нормализовать combined_features\nобученным Scaler'ом;
else (<b>False (Инференс)</b>)
  :Нормализовать combined_features\nзагруженным Scaler'ом;
endif

:Создать FraudDetectionInput\nдля каждого документа;

note right
  Поля FraudDetectionInput:
  • raw_text — очищенный текст
  • report_id — идентификатор
  • label — метка (если есть)
  • tfidf_features — вектор TF-IDF
  • linguistic_features — словарь метрик
  • combined_features — нормализованный вектор
end note

:Вернуть список FraudDetectionInput;

stop

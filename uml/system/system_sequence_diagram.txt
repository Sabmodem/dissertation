' ============================================================
' TRAIN PIPELINE (Обучение модели)
' ============================================================
== TrainPipeline: Обучение модели ==

actor "ML Developer" as Dev
participant "Experiment" as Exp
participant "DataLoader" as DL
participant "BaseVectorizer\n(Preparator)" as Vec
participant "BaseModel\n(Trainer)" as Model
collections "Файловая\nсистема" as FS

Dev -> Exp : run(data_file)
activate Exp

' Шаг 1: Загрузка данных
Exp -> DL : load_data(data_file)
activate DL
DL -> DL : pd.read_csv(filepath)
DL --> Exp : self
deactivate DL

' Шаг 2: Разбиение на выборки
Exp -> DL : split_data()
activate DL
DL -> DL : train_test_split(test_size=0.2)
DL -> DL : train_test_split(val_size=0.25)
note right of DL
    Получаем три выборки:
    • train_df (60%)
    • val_df (20%)
    • test_df (20%)
end note
DL --> Exp : self
deactivate DL

' Шаг 3: Предобработка
Exp -> DL : preprocess()
activate DL

' Шаг 3a: Кодирование меток
DL -> DL : LabelEncoder.fit_transform(train_df[label])
DL -> DL : LabelEncoder.transform(val_df[label])
DL -> DL : LabelEncoder.transform(test_df[label])
note right of DL
    'yes'/'no' → 1/0
end note

' Шаг 3b: Векторизация
DL -> Vec : fit_transform(train_df[text])
activate Vec
Vec -> Vec : fit(train_texts)\n(обучение словаря/токенизатора)
Vec -> Vec : transform(train_texts)\n(векторизация)
Vec --> DL : X_train: np.ndarray
deactivate Vec

DL -> Vec : transform(val_df[text])
activate Vec
Vec --> DL : X_val: np.ndarray
deactivate Vec

DL -> Vec : transform(test_df[text])
activate Vec
Vec --> DL : X_test: np.ndarray
deactivate Vec

DL --> Exp : self
deactivate DL

' Шаг 4: Получение данных
Exp -> DL : get_train_data()
activate DL
DL --> Exp : X_train, y_train
deactivate DL

Exp -> DL : get_val_data()
activate DL
DL --> Exp : X_val, y_val
deactivate DL

' Шаг 5: Обучение модели
Exp -> Model : train(X_train, y_train, X_val, y_val)
activate Model

alt SklearnModel
    Model -> Model : build_model()\n(уже инициализирован)
    Model -> Model : estimator.fit(X_train, y_train)
    note right of Model
        train_score = estimator.score(X_train)
        val_score = estimator.score(X_val)
    end note
    
else KerasModel
    Model -> Model : build_model()\n(Sequential + Dense + Dropout)
    note right of Model
        Архитектура: Input → [256, 128, 64] → Output
        Компиляция: optimizer, loss, metrics
    end note
    Model -> Model : model.fit(X_train, y_train,\nvalidation_data=(X_val, y_val),\ncallbacks=[EarlyStopping])
    note right of Model
        До 20 эпох
        EarlyStopping: patience=3
        Сохраняет лучшие веса
    end note
    
else HuggingFaceModel
    Model -> Model : build_model()\n(AutoModel + AutoTokenizer)
    note right of Model
        Загрузка pretrained модели
        Добавление pad_token если нужно
        Resize embeddings
    end note
    Model -> Model : Trainer.train()
    note right of Model
        Создаёт HFDataset для train и val
        Настраивает TrainingArguments
        Compute metrics: accuracy, f1, precision, recall
    end note
end

Model -> Model : Сохранение истории обучения
Model --> Exp : training_history: Dict
deactivate Model

' Шаг 6: Сохранение модели
Exp -> Model : save_model(filepath)
activate Model

alt SklearnModel
    Model -> FS : joblib.dump(model, filepath)
    note right of FS
        model.pkl
    end note
    
else KerasModel
    Model -> FS : model.save(filepath)
    note right of FS
        model.h5 / model.keras
    end note
    
else HuggingFaceModel
    Model -> FS : model.save_pretrained(filepath)
    Model -> FS : tokenizer.save_pretrained(filepath)
    note right of FS
        config.json + pytorch_model.bin
        tokenizer.json + vocab.json
    end note
end

Model --> Exp : void
deactivate Model

' Шаг 7: Оценка на тренировочной выборке (опционально)
opt evaluate_on_train
    Exp -> DL : get_train_data()
    activate DL
    DL --> Exp : X_train, y_train
    deactivate DL
    
    Exp -> Exp : Evaluator.evaluate(model, X_train, y_train, "train")
    note right of Exp
        Вычисление метрик:
        accuracy, precision, recall, f1
        confusion_matrix
        ROC curve, PR curve
    end note
end

' Шаг 8: Оценка на валидационной выборке
opt evaluate_on_val
    Exp -> DL : get_val_data()
    activate DL
    DL --> Exp : X_val, y_val
    deactivate DL
    
    Exp -> Exp : Evaluator.evaluate(model, X_val, y_val, "validation")
end

' Шаг 9: Оценка на тестовой выборке
opt evaluate_on_test
    Exp -> DL : get_test_data()
    activate DL
    DL --> Exp : X_test, y_test
    deactivate DL
    
    Exp -> Exp : Evaluator.evaluate(model, X_test, y_test, "test")
end

' Шаг 10: Визуализация
opt generate_plots
    loop для каждого dataset (train, validation, test)
        Exp -> Exp : Visualizer.plot_all(metrics, model_name)
        note right of Exp
            • Confusion Matrix
            • ROC Curve
            • Precision-Recall Curve
        end note
    end
    
    opt если есть training_history
        Exp -> Exp : Visualizer.plot_training_history(history)
        note right of Exp
            Графики loss и accuracy
            по эпохам
        end note
    end
end

Exp --> Dev : results: Dict
deactivate Exp

note right of Dev
    <b>Результат run():</b>
    {
      'train': {metrics},
      'validation': {metrics},
      'test': {metrics}
    }
    
    + файл модели на диске
    + графики в save_dir
end note

' Шаг 1: Загрузка внешних данных

DL2 -> DL2 : pd.read_csv(external_file)
DL2 -> DL2 : LabelEncoder.transform(df[label])

' Шаг 2: Инференс (Processor)
Exp2 -> Model2 : predict(X_external)
activate Model2

alt SklearnModel
    Model2 -> Model2 : estimator.predict(X)
    note right of Model2
        Возвращает метки классов [0, 1, 0, ...]
    end note
    
else KerasModel
    Model2 -> Model2 : model.predict(X, verbose=0)
    Model2 -> Model2 : (probabilities > 0.5).astype(int)
    note right of Model2
        Порог классификации: 0.5
    end note
    
else HuggingFaceModel
    Model2 -> Model2 : trainer.predict(dataset)
    Model2 -> Model2 : np.argmax(logits, axis=1)
    note right of Model2
        Конвертация logits → метки
    end note
end

Model2 --> Exp2 : y_pred: np.ndarray
deactivate Model2

' Шаг 3: Получение вероятностей
Exp2 -> Model2 : predict_proba(X_external)
activate Model2

alt SklearnModel
    Model2 -> Model2 : estimator.predict_proba(X)
    note right of Model2
        Если нет predict_proba:
        • SVM: decision_function + expit
        • Возвращает [P(0), P(1)]
    end note
    
else KerasModel
    Model2 -> Model2 : model.predict(X)
    Model2 -> Model2 : np.vstack([1-p, p]).T
    note right of Model2
        Формирует матрицу вероятностей
    end note
    
else HuggingFaceModel
    Model2 -> Model2 : trainer.predict(dataset)
    Model2 -> Model2 : softmax(logits)
    note right of Model2
        Конвертация logits → probabilities
    end note
end

Model2 --> Exp2 : y_proba: np.ndarray
deactivate Model2

    • ROC curve (fpr, tpr, auc)
    • PR curve (precision, recall)
    
    Результат упаковывается в FraudOutput
end note

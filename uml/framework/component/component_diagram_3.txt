!theme plain

skinparam componentBorderColor DarkSlateGray

package "Пользовательские Реализации" as User {
  component "MyPreparator" as MyPrep
  component "MyProcessor" as MyProc
  component "MyInterpreter" as MyInterp
  component "MyTrainer" as MyTrain
}

package "Внешние Интерфейсы" as Ext {
  component "Raw Data\n(CSV, JSON, Images...)" as Raw
  component "File System\n(model.pkl, model.keras...)" as FS
  database "External Systems\n(DB, APIs, Alerts)" as ExtSys
}

' ========= Внешние зависимости =========
MyProc ..> FS : <<читает модель>>
MyTrain ..> FS : <<записывает модель>>

Raw ..> MyPrep : вход
ExtSys <.. MyInterp : выход

note bottom of Ext
  <b>Внешние зависимости</b>
  Фреймворк не управляет форматами файлов
  и внешними системами.
  Пользователь сам реализует чтение/запись.
end note

note top of MyPrep
  prepare(rawData: Any)
  → List<MyInput>
  
  Читает сырые данные,
  преобразует в список DTO.
end note

note top of MyProc
  process(input: List<MyInput>)
  → List<MyOutput>
  
  Загружает модель из файла
  один раз для всего списка.
  Обрабатывает все элементы.
end note

note top of MyTrain
  train(input: List<MyInput>)
  → void
  
  Обучает модель на всём
  списке данных.
  Сохраняет модель в файл.
end note

note top of MyInterp
  interpret(output: List<MyOutput>)
  → Any
  
  Получает все результаты сразу
  для агрегированной интерпретации
  (гистограммы, сводки, алерты).
end note

@enduml

@startuml
!theme plain
skinparam participantBackgroundColor LightYellow
skinparam participantBorderColor DarkSlateGray
skinparam sequenceArrowColor DarkSlateGray

actor "Исследователь" as Dev
participant "InferencePipeline" as Inf
participant "TextFraudPreparator" as Prep
participant "RFProcessor" as RFProc
participant "RFInterpreter" as RFInterp
collections "Файловая\nсистема" as FS

Dev -> Inf : run(rawData)
activate Inf

' ========= Этап подготовки =========
Inf -> Prep : prepare(rawData)
activate Prep

note right of Prep
  <b>fit_mode = False:</b>
  загружает TfidfVectorizer и Scaler
  с диска, применяет к новым данным.
end note

Prep -> FS : загружает артефакты\n(TfidfVectorizer, StandardScaler)

Prep --> Inf : List[FraudDetectionInput]\n(combined_features)
deactivate Prep

' ========= Цикл инференса =========
loop для каждого FraudDetectionInput

Inf -> RFProc : process(FraudDetectionInput)
activate RFProc

RFProc -> FS : загружает артефакты\n(rf_model.joblib,\ntfidf_vectorizer.pkl,\nfeature_scaler.pkl,\nfeature_names.json)

note right of RFProc
  <b>Инференс RF</b>
  1. Извлекает combined_features.
  2. model.predict_proba() →
     вероятность fraud.
  3. model.predict() → метка.
  4. Формирует FraudDetectionOutput:
     • prediction
     • fraud_probability
     • method_name = "RandomForest"
end note

RFProc --> Inf : FraudDetectionOutput
deactivate RFProc

' ========= Интерпретация =========
Inf -> RFInterp : interpret(FraudDetectionInput,\nFraudDetectionOutput)
activate RFInterp

note right of RFInterp
  <b>SHAP-интерпретация</b>
  1. Вычисляет SHAP-значения
     (TreeExplainer).
  2. Извлекает топ-K признаков,
     наиболее повлиявших на решение.
  3. Формирует explanation:
     {"feature_name": shap_value, ...}
  4. Обогащает FraudDetectionOutput
     полем explanation.
  5. Форматирует читаемый отчёт.
end note

RFInterp --> Inf : Any (отчёт с SHAP-объяснениями)
deactivate RFInterp

end

Inf --> Dev : Any (сводный результат)
deactivate Inf

@enduml

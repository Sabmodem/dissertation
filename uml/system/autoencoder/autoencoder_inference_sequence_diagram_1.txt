@startuml
!theme plain
skinparam participantBackgroundColor LightYellow
skinparam participantBorderColor DarkSlateGray
skinparam sequenceArrowColor DarkSlateGray

actor "Исследователь" as Dev
participant "InferencePipeline" as Inf
participant "TextFraudPreparator" as Prep
participant "AEProcessor" as AEProc
collections "Файловая\nсистема" as FS

Dev -> Inf : run(rawData)
activate Inf

Inf -> Prep : prepare(rawData)
activate Prep

note right of Prep
  <b>fit_mode = False</b>
  Загружает TfidfVectorizer и Scaler,
  применяет к новым данным.
  Возвращает FraudDetectionInput
  с combined_features (без меток).
end note

Prep -> FS : загружает артефакты\n(TfidfVectorizer, StandardScaler)

Prep --> Inf : List[FraudDetectionInput]
deactivate Prep

loop для каждого FraudDetectionInput

  Inf -> AEProc : process(FraudDetectionInput)
  activate AEProc

  AEProc -> FS : загружает артефакты\n(autoencoder_model.h5,\nanomaly_threshold.json,\nfeature_scaler.pkl)

  note right of AEProc
    <b>Инференс AE</b>
    1. Загружает модель и порог
    2. Извлекает combined_features
    3. model.predict() → восстановленный вектор
    4. MSE = средний квадрат ошибки
    5. anomaly_score = MSE / threshold
    6. Если MSE > threshold → prediction = 1
    7. Формирует FraudDetectionOutput:
       prediction, anomaly_score,
       method_name = "Autoencoder",
       explanation = {mse, threshold}
  end note

  AEProc --> Inf : FraudDetectionOutput
  deactivate AEProc

end

@enduml
